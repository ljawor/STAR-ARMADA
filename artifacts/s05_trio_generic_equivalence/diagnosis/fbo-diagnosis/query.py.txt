import json
from pathlib import Path
import pygame
from OpenGL import GL
out=Path(__file__).parent
pygame.display.init()
pygame.display.gl_set_attribute(pygame.GL_MULTISAMPLEBUFFERS,1)
pygame.display.gl_set_attribute(pygame.GL_MULTISAMPLESAMPLES,4)
pygame.display.set_mode((1280,800),pygame.OPENGL|pygame.DOUBLEBUF)
keys=['RED_BITS','GREEN_BITS','BLUE_BITS','ALPHA_BITS','DEPTH_BITS','STENCIL_BITS','SAMPLES','SAMPLE_BUFFERS','READ_BUFFER','DRAW_BUFFER']
r={k:int(GL.glGetIntegerv(getattr(GL,'GL_'+k))) for k in keys}
r.update({k:GL.glGetString(getattr(GL,'GL_'+k)).decode() for k in ['VENDOR','RENDERER','VERSION','EXTENSIONS']})
r['dither']=bool(GL.glIsEnabled(GL.GL_DITHER))
assert 'GL_ARB_framebuffer_object' in r['EXTENSIONS'].split()
from OpenGL.GL.ARB import framebuffer_object as arb
assert bool(arb.glGetFramebufferAttachmentParameteriv)
r['attachment_query_entrypoint']=True
r['attachments']={name:{key:int(arb.glGetFramebufferAttachmentParameteriv(arb.GL_FRAMEBUFFER,attachment,getattr(arb,'GL_FRAMEBUFFER_ATTACHMENT_'+key))) for key in ['COMPONENT_TYPE','COLOR_ENCODING']} for name,attachment in [('color',GL.GL_BACK_LEFT),('depth',GL.GL_DEPTH)]}
from OpenGL.GL.EXT import framebuffer_object as f,framebuffer_multisample as m,framebuffer_blit as b
r['entrypoints']={name:bool(getattr(mod,name)) for mod,name in [(f,'glBindFramebufferEXT'),(f,'glGetRenderbufferParameterivEXT'),(m,'glRenderbufferStorageMultisampleEXT'),(b,'glBlitFramebufferEXT')]}
if 'GL_EXT_framebuffer_multisample' in r['EXTENSIONS'].split():r['MAX_SAMPLES_EXT']=int(GL.glGetIntegerv(m.GL_MAX_SAMPLES_EXT))
(out/'driver-query.json').write_text(json.dumps(r,indent=2)); print(json.dumps(r,indent=2));pygame.display.quit()
